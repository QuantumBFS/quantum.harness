w1 = XOR x1 x5
w2 = AND x1 x5
w3 = XOR x2 x6
w4 = XOR w3 w2
w5 = AND x2 x6
w6 = AND w3 w2
w7 = OR w5 w6
w8 = XOR x3 x7
w9 = XOR w8 w7
w10 = AND x3 x7
w11 = AND w8 w7
w12 = OR w10 w11
w13 = XOR x4 x8
w14 = XOR w13 w12
w15 = AND x4 x8
w16 = AND w13 w12
w17 = OR w15 w16
OUTPUTS w1 w4 w9 w14 w17
