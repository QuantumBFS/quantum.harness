> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build and freeze an exact-LLL analytic `rho_bar(2,M)|Laughlin>` multiplet dressed by a trainable SO(3)-scalar projected-density operator network.

**Architecture:** The route evaluates the Laughlin polynomial directly on sphere spinors and applies one-body LLL tensors through the finite monopole-orbital reproducing kernel, so flux and LLL closure never leave the fixed `Q` space. The five mother states are generated by the rank-2 projected density. A two-layer width-64 bank of scalar words built from `S_l=sum_m(-1)^m rho_bar(l,m)rho_bar(l,-m)`, `l=2,3,4`, acts on both the `L=0` parent and all `L=2` components; only scalar coefficients are trained.

**Tech Stack:** Python 3.11+, NumPy, SciPy, SymPy for cached Wigner/CG constants, pytest, JSON, Git worktrees.

```python
def laughlin_amplitude(spinors, power=3):
    value = 1.0 + 0.0j
    for i in range(len(spinors)):
        for j in range(i + 1, len(spinors)):
            ui, vi = spinors[i]
            uj, vj = spinors[j]
            value *= (ui * vj - uj * vi) ** power
    return value

@functools.lru_cache(maxsize=None)
def density_tensor_matrix(two_q: int, rank: int, component: int) -> np.ndarray:
    q = Rational(two_q, 2)
    matrix = np.zeros((two_q + 1, two_q + 1), dtype=np.complex128)
    for a in range(two_q + 1):
        ma = Rational(-two_q + 2*a, 2)
        for b in range(two_q + 1):
            mb = Rational(-two_q + 2*b, 2)
            matrix[a,b] = complex(((-1)**int(q-ma)) * sqrt(2*rank+1) * wigner_3j(q, rank, q, -ma, component, mb))
    return matrix
```

```python
def apply_one_body_tensor(psi, spinors, tensor, grid):
    total = 0.0j
    orbitals_here = monopole_vector(spinors, grid.two_q)
    for particle in range(len(spinors)):
        replaced = np.repeat(spinors[None, :, :], len(grid.spinors), axis=0)
        replaced[:, particle, :] = grid.spinors
        coefficients = grid.weights[:, None] * grid.orbitals.conj() * np.asarray([psi(x) for x in replaced])[:, None]
        projected = coefficients.sum(axis=0)
        total += orbitals_here[particle] @ tensor @ projected
    return total
```

def scalar_density_action(amplitude, spinors, rank, grid):
    total = 0.0j
    for m in range(-rank, rank + 1):
        first = lambda z, m=m: apply_one_body_tensor(amplitude, z, density_tensor_matrix(grid.two_q, rank, -m), grid)
        total += (-1)**m * apply_one_body_tensor(first, spinors, density_tensor_matrix(grid.two_q, rank, m), grid)
    return total

- [ ] **Step 3: Implement the two-layer width-64 word bank**

```python
def word_amplitude(self, mother, spinors, hidden):
    state = mother
    for layer in range(2):
        previous = state
        coefficients = self.trunk[layer, hidden]
        state = lambda z, previous=previous, coefficients=coefficients: (
            previous(z) + sum(coefficients[r] * scalar_density_action(previous, z, rank, self.grid) for r, rank in enumerate((2,3,4)))
        )
    return state(spinors)

def amplitude(self, spinors, sector, m=0):
    mother = self.mothers.ground if sector == "ground" else self.mothers.component(m)
    head = self.ground_head if sector == "ground" else self.excited_head
    return mother(spinors) + sum(head[h] * self.word_amplitude(mother, spinors, h) for h in range(64))
```

Trunk parameters are shared; only two scalar heads differ by sector, and the same excited head is used for all `M`. Initialization is identity plus `1e-4` complex coefficients so the exact seed is the zero-parameter limit.

class AnalyticState:
    def sample(self, n_samples, seed):
        configs = metropolis_spinors(self.log_amplitude_one, n_samples=n_samples, burn_in=self.protocol.sampling["burn_in_steps"], seed=seed)
        return SampleBatch(configs, n_samples, self.protocol.sampling["burn_in_steps"], seed)

    def local_energy(self, configs):
        return np.asarray([chord_coulomb(z, q=self.two_q/2) for z in configs], dtype=np.complex128)

        return np.asarray([casimir_ratio(self.amplitude_one, z) for z in configs], dtype=np.complex128)
